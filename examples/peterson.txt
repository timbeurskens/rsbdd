"
Model checking Peterson's mutual exclusion protocol for two processes P0 and P1.

Each process cycles through idle -> wait -> crit -> idle:

    idle: raise the flag and give the turn to the other process
    wait: enter the critical section if the other process is idle or it is our turn
    crit: lower the flag and leave

The state is a function of w: the current state Cur, its predecessor Prev or its successor Next:

    pc(w, p, l)   the program counter of process p is l   (Idle, Wait or Crit)
    turn(w)       turn (false = P0, true = P1)

The flag of a process is raised exactly when it is not idle, so it is not a separate variable.

Checked properties:

    mutex          no reachable state has both processes in the critical section      (lfp)
    deadlock_free  every reachable state has a successor
    recoverable    from every reachable state, P0 can still reach its critical section (lfp)
    no_overtaking  while P0 waits, P1 cannot keep moving forever on its own             (gfp)

Run with:
    rsbdd examples/peterson.txt -t
"

let init := is_at(Cur, P0, Idle) & is_at(Cur, P1, Idle) & -turn(Cur);

let Reach := lfp R # init |
    exists pc(Prev, _, _), turn(Prev) #
        R[pc(Cur, P0, _) := pc(Prev, P0, _), pc(Cur, P1, _) := pc(Prev, P1, _), turn(Cur) := turn(Prev)] & step(Prev, Cur);

let CanCrit0 := lfp Z # is_at(Cur, P0, Crit) |
    exists pc(Next, _, _), turn(Next) #
        step(Cur, Next) & Z[pc(Cur, P0, _) := pc(Next, P0, _), pc(Cur, P1, _) := pc(Next, P1, _), turn(Cur) := turn(Next)];

let Overtaken := gfp Z # is_at(Cur, P0, Wait) &
    exists pc(Next, _, _), turn(Next) #
        step1(Cur, Next) & keep(P0, Cur, Next) & Z[pc(Cur, P0, _) := pc(Next, P0, _), pc(Cur, P1, _) := pc(Next, P1, _), turn(Cur) := turn(Next)];

location(Idle) -> true;
location(Wait) -> true;
location(Crit) -> true;

"In state w the program counter of process p is l, and no other location"
sum w, p, l # is_at(w, p, l) -> forall k # location(k) => (pc(w, p, k) <=> k = l);

"The program counter of process p, or the turn, does not change from state u to v"
sum p, u, v # keep(p, u, v) -> forall k # location(k) => (pc(v, p, k) <=> pc(u, p, k));
sum u, v # keept(u, v) -> (turn(v) <=> turn(u));

"A step of P0 or P1 from state u to state v"
sum u, v # step0(u, v) ->
    is_at(u, P0, Idle) & is_at(v, P0, Wait) & turn(v) |
    is_at(u, P0, Wait) & (is_at(u, P1, Idle) | -turn(u)) & is_at(v, P0, Crit) & keept(u, v) |
    is_at(u, P0, Crit) & is_at(v, P0, Idle) & keept(u, v);

sum u, v # step1(u, v) ->
    is_at(u, P1, Idle) & is_at(v, P1, Wait) & -turn(v) |
    is_at(u, P1, Wait) & (is_at(u, P0, Idle) | turn(u)) & is_at(v, P1, Crit) & keept(u, v) |
    is_at(u, P1, Crit) & is_at(v, P1, Idle) & keept(u, v);

sum u, v # step(u, v) -> step0(u, v) & keep(P1, u, v) | step1(u, v) & keep(P0, u, v);

mutex <=> forall pc(Cur, _, _), turn(Cur) # Reach => -(is_at(Cur, P0, Crit) & is_at(Cur, P1, Crit));
deadlock_free <=> forall pc(Cur, _, _), turn(Cur) # Reach => exists pc(Next, _, _), turn(Next) # step(Cur, Next);
recoverable <=> forall pc(Cur, _, _), turn(Cur) # Reach => CanCrit0;
no_overtaking <=> forall pc(Cur, _, _), turn(Cur) # Reach => -Overtaken
