"
Which rooms can be reached from the hall? Rooms and doors are terms defined by rewrite rules,
reachability is a least fixpoint over a 3-bit encoding of the rooms.

Run with:
    rsbdd examples/rooms.txt -t
"

let T := exists x, y # step(x, y) & at(x)[b2 := p2, b1 := p1, b0 := p0] & at(y);
let Reach := lfp R # at(Hall) | exists p2, p1, p0 # R[b2 := p2, b1 := p1, b0 := p0] & T;

room(Hall) -> true;
room(Kitchen) -> true;
room(Garden) -> true;
room(Shed) -> true;
room(Attic) -> true;
room(Cellar) -> true;

door(Hall, Kitchen) -> true;
door(Kitchen, Garden) -> true;
door(Garden, Shed) -> true;
door(Attic, Cellar) -> true;
trapdoor(Attic, Hall) -> true;

sum x, y # step(x, y) -> door(x, y) | door(y, x) | trapdoor(x, y);

at(Hall)    -> -b2 & -b1 & -b0;
at(Kitchen) -> -b2 & -b1 &  b0;
at(Garden)  -> -b2 &  b1 & -b0;
at(Shed)    -> -b2 &  b1 &  b0;
at(Attic)   ->  b2 & -b1 & -b0;
at(Cellar)  ->  b2 & -b1 &  b0;

forall r # room(r) => (reachable(r) <=> exists b2, b1, b0 # at(r) & Reach)
