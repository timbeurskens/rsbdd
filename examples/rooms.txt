"
Which rooms of a house can be reached from the hall? Rooms and doors are terms defined by rewrite rules,
reachability is a least fixpoint over the location of a visitor.

The location is a function: the unknowns at(w, r) state that in state w the visitor is in room r. The state Cur is
the current location, Prev the location before the last step. The substitution R[at(Cur, _) := at(Prev, _)]
renames the current location in R to the previous one.

Run with:
    rsbdd examples/rooms.txt -t
"

let T := exists x, y # step(x, y) & is_at(Prev, x) & is_at(Cur, y);
let Reach := lfp R # is_at(Cur, Hall) | exists at(Prev, _) # R[at(Cur, _) := at(Prev, _)] & T;

room(Hall) -> true;
room(Kitchen) -> true;
room(Garden) -> true;
room(Shed) -> true;
room(Attic) -> true;
room(Cellar) -> true;

door(Hall, Kitchen) -> true;
door(Kitchen, Garden) -> true;
door(Garden, Shed) -> true;
door(Attic, Cellar) -> true;
trapdoor(Attic, Hall) -> true;

sum x, y # step(x, y) -> door(x, y) | door(y, x) | trapdoor(x, y);

"In state w the visitor is in room r, and in no other room"
sum w, r # is_at(w, r) -> forall q # room(q) => (at(w, q) <=> q = r);

forall r # room(r) => (reachable(r) <=> exists at(Cur, _) # is_at(Cur, r) & Reach)
